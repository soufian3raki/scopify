ingredients:
  - name: Bread flour
    amount: 500g
  - name: Milk (cold)
    amount: 230g
  - name: Active starter
    amount: 100g
  - name: Sugar
    amount: 55g
  - name: Salt
    amount: 10g
  - name: Butter (lamination block)
    amount: 280g
steps:
  - text: Mix firm dough and ferment.
    time: ⏱ 12 hours
    cookTitle: Dough
    cookText: Mix a firm dough and bulk ferment overnight in the fridge to develop slowly.
    cookTime: ⏱ 12 hours
  - text: Prepare the butter block.
    time: ⏱ 30 min
    cookTitle: Butter
    cookText: Pound chilled butter into a flat square at 16°C — it must be pliable, not greasy.
    cookTime: ⏱ 30 minutes
  - text: Lock and laminate.
    time: ⏱ 1.5 hours
    cookTitle: Laminate
    cookText: Encase the butter, then perform 3 letter folds, chilling 30 min between each.
    cookTime: ⏱ 1.5 hours
  - text: Shape and proof.
    time: ⏱ 4 hours
    cookTitle: Shape
    cookText: Roll thin, cut triangles, and roll up. Proof 3–4 h until jiggly.
    cookTime: ⏱ 4 hours
  - text: Bake until flaky.
    time: ⏱ 20 min
    cookTitle: Bake
    cookText: Egg-wash and bake at 200°C for ~20 min until deeply golden.
    cookTime: ⏱ 20 minutes
tips:
  - icon: 🌡️
    title: Temperature
    text: If butter softens or breaks through, chill the dough immediately and continue.
  - icon: 🌬️
    title: Proof Fully
    text: Under-proofed croissants leak butter; they should wobble when tapped.
