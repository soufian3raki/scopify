difficulty: Easy
times:
  prep: 10 min
  rest: 5 min
  bake: 15 min
  total: 30 min
yield: 8 pancakes
rating: "4.7"
ingredients:
  - name: Sourdough discard
    amount: 200g
  - name: Whole milk
    amount: 120g
  - name: Egg (1 large)
    amount: 50g
  - name: Plain flour
    amount: 80g
  - name: Sugar
    amount: 15g
  - name: Baking soda
    amount: 3g
  - name: Fine salt
    amount: 2g
  - name: Melted butter
    amount: 20g
steps:
  - text: Mix wet ingredients.
    time: ⏱ 5 min
    cookTitle: Wet Mix
    cookText: Whisk discard, milk, egg and melted butter until smooth.
    cookTime: ⏱ 5 minutes
  - text: Incorporate dry ingredients and rest.
    time: ⏱ 5 min
    cookTitle: Combine
    cookText: Fold in flour, sugar, salt and baking soda. Rest for 5 minutes until it bubbles.
    cookTime: ⏱ 5 minutes
  - text: Cook on a medium pan.
    time: ⏱ 15 min
    cookTitle: Cook
    cookText: Ladle onto the pan. Flip when bubbles set on top, approx. 2 min per side.
    cookTime: ⏱ 15 minutes
tips:
  - icon: 🍳
    title: Heat
    text: Cook on medium, not high — the sugar in the discard browns fast.
  - icon: 🥣
    title: Texture
    text: Do not overmix the batter; a few lumps are perfectly fine.
